import tkinter as tk

def key_pressed(event):
    key = event.keysym

    with open("keystrokes.txt", "a") as file:
        file.write(key + "\n")

    label.config(text=f"Last Key Pressed: {key}")

root = tk.Tk()
root.title("Keystroke Visualizer")
root.geometry("400x200")

label = tk.Label(root, text="Press any key inside this window", font=("Arial", 12))
label.pack(pady=50)

root.bind("<Key>", key_pressed)

root.mainloop()
